26struct BBox {
27 float x1 = 0;
28 float y1 = 0;
29 float x2 = 0;
30 float y2 = 0;
31
32 // Constructors
33 BBox() = default;
34 BBox(float x1, float y1, float x2, float y2)
35 : x1(x1), y1(y1), x2(x2), y2(y2) {}
36
37 // Properties
38 float width() const { return x2 - x1; }
39 float height() const { return y2 - y1; }
40 float area() const { return width() * height(); }
41 float center_x() const { return (x1 + x2) / 2; }
42 float center_y() const { return (y1 + y2) / 2; }
43
47 float iou(const BBox& other) const {
48 float inter_x1 = std::max(x1, other.x1);
49 float inter_y1 = std::max(y1, other.y1);
50 float inter_x2 = std::min(x2, other.x2);
51 float inter_y2 = std::min(y2, other.y2);
52
53 float inter_w = std::max(0.0f, inter_x2 - inter_x1);
54 float inter_h = std::max(0.0f, inter_y2 - inter_y1);
55 float inter_area = inter_w * inter_h;
56
57 float union_area = area() + other.area() - inter_area;
58 return union_area > 0 ? inter_area / union_area : 0;
59 }
60
64 std::array<float, 4> to_xywh() const {
65 return {x1, y1, width(), height()};
66 }
67
71 std::array<float, 4> to_cxcywh() const {
72 return {center_x(), center_y(), width(), height()};
73 }
74
78 cv::Rect to_rect() const {
79 return cv::Rect(
80 static_cast<int>(x1),
81 static_cast<int>(y1),
82 static_cast<int>(width()),
83 static_cast<int>(height())
84 );
85 }
86
90 static BBox from_xywh(float x, float y, float w, float h) {
91 return BBox(x, y, x + w, y + h);
92 }
93
97 static BBox from_cxcywh(float cx, float cy, float w, float h) {
98 return BBox(cx - w/2, cy - h/2, cx + w/2, cy + h/2);
99 }
100};
